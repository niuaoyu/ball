  const BRIDGE = 'http://127.0.0.1:8321';

  const PANEL_HTML = `

  const CSS = `

    display:flex;justify-content:space-between;align-items:center}

  #wbclip-panel .wbclip-row{display:flex;gap:6px;flex-wrap:wrap;margin-bottom:8px}

    background:#f5f3ee;border-radius:7px;cursor:pointer;color:#222}
  #wbclip-panel button:hover{background:#ece9e1}
  #wbclip-panel button.primary{background:#0f766e;border-color:#0f766e;color:#fff}
  #wbclip-panel button.on{background:#0f766e;border-color:#0f766e;color:#fff}
  #wbclip-panel button.fullbtn{width:100%;background:#fef2f2;border-color:#fca5a5;color:#b91c1c}

  #wbclip-panel button b{font-weight:700;opacity:.75}

  #wbclip-panel .wbclip-txdur{font-size:12px;color:#0f766e;margin-bottom:6px;font-variant-numeric:tabular-nums}
  #wbclip-panel input{width:100%;font:inherit;padding:6px 9px;border:1px solid #d8d4ca;
    border-radius:7px;margin-bottom:6px;outline:none;background:#faf9f5}
  #wbclip-panel input:focus{border-color:#0f766e;background:#fff}
  #wbclip-progress{font-size:12px;color:#8a857a;min-height:16px;word-break:break-all}

  const styleEl = document.createElement('style');
  styleEl.textContent = CSS;
  document.head.appendChild(styleEl);
  const box = document.createElement('div');
  box.innerHTML = PANEL_HTML;
  document.body.appendChild(box.firstElementChild);

  const $ = s => document.querySelector(s);
  const panel = $('#wbclip-panel');
  const el = {
    state: $('#wbclip-state'),
    a: $('#wbclip-a'), b: $('#wbclip-b'), now: $('#wbclip-now'),
    back5: $('#wbclip-back5'), fwd5: $('#wbclip-fwd5'),
    full: $('#wbclip-full'), fullrow: $('#wbclip-fullrow'),
    tags: $('#wbclip-tags'), note: $('#wbclip-note'),
    txrow: $('#wbclip-txrow'), start: $('#wbclip-start'), end: $('#wbclip-end'), dur: $('#wbclip-dur'),
    progress: $('#wbclip-progress'), preview: $('#wbclip-preview'),
    undo: $('#wbclip-undo')
  };

  const S = { video: null, A: null, B: null, busy: false, previewing: false, lastQueuedId: null };
  el.undo.style.display = 'none'; // 撤回按钮默认隐藏，入队成功后才显示

  /* 解析时间：用空格分隔「时 分 秒」，缺省默认 0，从右往左对齐。
     55 → 55 秒；5 30 → 5 分 30 秒；1 02 00 → 1 小时 2 分 0 秒 */
  function parseTime(str) {
    const s = (str || '').trim();
    if (!s) return null;
    if (!/^\d+(\s+\d{1,2}){0,2}$/.test(s)) return null;
    const parts = s.split(/\s+/).map(Number);
    let sec = 0;
    for (let i = 0; i < parts.length; i++) sec = sec * 60 + parts[i];
    return sec;
  }
  function fmtDur(sec) {
    if (sec == null || !isFinite(sec) || sec < 0) return '--';
    const m = Math.floor(sec / 60), s = Math.round(sec % 60);
    if (m === 0) return s + ' 秒';
    if (m < 60) return m + ' 分 ' + s + ' 秒';
    const h = Math.floor(m / 60);
    return h + ' 小时 ' + (m % 60) + ' 分 ' + s + ' 秒';
  }

  function fmt(t) {
    if (t == null || !isFinite(t) || t < 0) return '--:--.---';
    const m = Math.floor(t / 60), s = Math.floor(t % 60), ms = Math.floor((t % 1) * 1000);
    return String(m).padStart(2, '0') + ':' + String(s).padStart(2, '0') + '.' + String(ms).padStart(3, '0');
  }
  const r3 = n => Math.round(n * 1000) / 1000;
  function uid(p) {
    return p + '-' + Date.now().toString(36) + '-' + Math.random().toString(36).slice(2, 8);
  }
  function setProgress(t) { el.progress.textContent = t || ''; }
  function renderTimes() {
    el.a.textContent = fmt(S.A);
    el.b.textContent = fmt(S.B);
    el.now.textContent = S.video ? fmt(S.video.currentTime) : '--:--.---';
  }
  function getFid() {
    const m = location.pathname.match(/show\/([0-9:]+)/) || location.search.match(/fid=([0-9:]+)/);
    return m ? m[1] : '';
  }

  function getSource() {
    const href = location.href;
    if (/weibo\.com|video\.weibo\.com/i.test(href)) {
      const fid = getFid();
      return fid ? { provider: 'weibo', source_url: href, fid } : null;
    }
    if (/^https?:\/\/v\.qq\.com\/x\/(?:cover|page)\//i.test(href)) {
      return { provider: 'tencent', source_url: href, fid: null };
    }
    return null;
  }

  // 当前是否为腾讯视频页
  function isTencentPage() {
    return /^https?:\/\/v\.qq\.com\//i.test(location.href);
  }

  function findVideo() {
    const vids = Array.from(document.querySelectorAll('video'))
      .filter(v => (v.currentSrc || v.src) && (v.readyState >= 1 || v.duration > 0));
    if (!vids.length) return null;
    if (isTencentPage()) {
      // 腾讯用 MSE 播 m3u8：正片播放器就是页面上那个 blob 流，video.duration 只反映
      // 「当前已缓冲的分段长度」（约 15s），永远读不到总时长。所以不能按 duration 挑，
      // 直接取页面上最大的那个 video 即可（通常只有一个，就是正片）。
      vids.sort((x, y) => (y.videoWidth || 0) - (x.videoWidth || 0));
      return vids[0];
    }
    // 微博是普通 mp4，duration 准确，取时长最长的那个。
    vids.sort((x, y) => (y.duration - x.duration) || ((y.videoWidth || 0) - (x.videoWidth || 0)));
    return vids[0];
  }

  let waitCount = 0;
  let lastVideo = null;
  const waitTimer = setInterval(() => {
    // 腾讯：播放器用 canvas 渲染、无 <video> 元素，改用手动填起止时间（下方两个输入框）
    if (isTencentPage()) {
      clearInterval(waitTimer);
      el.txrow.style.display = '';
      el.state.textContent = '手动填起止时间（空格分格 时 分 秒）';
      el.preview.style.display = 'none'; // 腾讯无 video，无法预览循环
      el.back5.style.display = 'none';
      el.fwd5.style.display = 'none';
      el.fullrow.style.display = ''; // 腾讯也支持完整视频下载（bridge 走分辨率选择器）
      return;
    }
    const v = findVideo();
    if (v) {
      if (v !== lastVideo) {
        S.video = v;
        lastVideo = v;
        S.A = null; S.B = null;
        renderTimes();
      }
      el.back5.style.display = '';
      el.fwd5.style.display = '';
      el.state.textContent = '就绪 · ' + Math.round(v.duration) + 's';
    } else if (++waitCount > 300) {
      clearInterval(waitTimer);
      el.state.textContent = '未找到可访问的视频播放器';
    }
  }, 800);
  setInterval(renderTimes, 120);

  /* 腾讯手动填时间的实时时长预览 */
  function refreshTxDur() {
    if (!isTencentPage()) return;
    const a = parseTime(el.start.value), b = parseTime(el.end.value);
    if (a == null || b == null) { el.dur.textContent = '时长：--'; return; }
    if (b <= a) { el.dur.textContent = '时长：结束须大于起始'; return; }
    el.dur.textContent = '时长：' + fmtDur(b - a);
  }

  function seekBy(delta) {
    if (isTencentPage()) return setProgress('腾讯页没有可读取的播放器时间，请手动填写时间');
    if (!S.video || S.busy) return setProgress('视频尚未就绪');
    stopPreview();
    const duration = isFinite(S.video.duration) && S.video.duration > 0 ? S.video.duration : Infinity;
    const next = Math.max(0, Math.min(duration, S.video.currentTime + delta));
    try { S.video.currentTime = next; } catch (_) { return setProgress('当前播放器不支持跳转'); }
    renderTimes();
  }

  function setA() {
    if (isTencentPage()) return setProgress('腾讯页请在下方输入框手动填起始时间');
    if (!S.video || S.busy) return;
    stopPreview();
    S.A = S.video.currentTime;
    if (S.B != null && S.B <= S.A) S.B = null;
    renderTimes();
  }
  function setB() {
    if (isTencentPage()) return setProgress('腾讯页请在下方输入框手动填结束时间');
    if (!S.video || S.busy) return;
    if (S.A == null) return setProgress('先按 I 标记开始点');
    S.B = S.video.currentTime;
    if (S.B <= S.A) { const t = S.A; S.A = S.B; S.B = t; }
    renderTimes();
  }
  function clearAll() {
    stopPreview();
    S.A = null; S.B = null;
    renderTimes(); setProgress('');
  }

  function togglePreview() {
    if (!S.video) return;
    if (S.previewing) return stopPreview();
    if (S.A == null || S.B == null) return setProgress('先标记 A 和 B 才能预览');
    S.video.currentTime = S.A;
    S.video.play();
    S.previewing = true;
    el.preview.classList.add('on');
    setProgress('预览循环中（按 P 停止）');
  }
  function stopPreview() {
    if (!S.previewing) return;
    S.previewing = false;
    el.preview.classList.remove('on');
    setProgress('');
  }
  function previewTick() {
    if (S.previewing && S.video && S.B != null && !S.video.paused) {
      if (S.video.currentTime >= S.B) S.video.currentTime = S.A;
    }
    requestAnimationFrame(previewTick);
  }
  requestAnimationFrame(previewTick);

  async function save() {
    if (S.busy) return;
    const source = getSource();
    if (!source) return setProgress('未能识别视频来源');

    let A, B;
    if (source.provider === 'tencent') {
      // 腾讯：从输入框读「时 分 秒」（空格分隔），不支持抓取（播放器无 video 元素）
      const a = parseTime(el.start.value), b = parseTime(el.end.value);
      if (a == null || b == null) return setProgress('请填起始/结束时间（如 55、5 30、1 02 00）');
      if (b <= a) return setProgress('结束时间须大于起始时间');
      A = a; B = b;
    } else {
      if (!S.video) return setProgress('视频未就绪');
      if (S.A == null || S.B == null) return setProgress('先用 I / O 标记起止点');
      A = Math.min(S.A, S.B); B = Math.max(S.A, S.B);
    }

    S.busy = true;
    el.state.textContent = '入队中…';
    stopPreview();
    try {
      const tagWords = el.tags.value.trim().split(/\s+/).filter(Boolean);
      // 标签不强制归类到「战术」，统一存为无维度纯标签，维度留空
      const tags = tagWords.map(v => ({ dimension: '', value: v }));
      const note = el.note.value.trim();
      const title = document.title.replace(/\s*-\s*微博.*$/, '').replace(/\s*-\s*腾讯.*$/, '').trim();
      const res = await postToBridge('/queue', {
        provider: source.provider,
        source_url: source.source_url,
        fid: source.fid,
        title,
        start_sec: r3(A), end_sec: r3(B), mode: 'clip',
        tags, note
      });
      if (!res || !res.ok) throw new Error((res && res.error) || '入队失败');
      S.lastQueuedId = res.id; // 记住刚入队这条，供「撤回」
      setProgress('已入队 ✓ 回电脑跑 node import.js --queue 下载裁剪');
      el.state.textContent = '已入队';
      el.undo.style.display = '';
      el.tags.value = ''; el.note.value = '';
      if (source.provider === 'tencent') { el.start.value = ''; el.end.value = ''; el.dur.textContent = '时长：--'; }
      clearAll();
    } catch (err) {
      setProgress('入队失败：' + err.message);
      el.state.textContent = '失败';
    } finally {
      S.busy = false;
    }
  }

  async function downloadFull() {
    if (S.busy) return;
    const source = getSource();
    if (!source) return setProgress('未能识别视频来源');
    S.busy = true;
    el.state.textContent = '请求下载…';
    const title = document.title.replace(/\s*-\s*微博.*$/, '').replace(/\s*-\s*腾讯.*$/, '').trim();
    try {
      const res = await postToBridge('/download', {
        url: source.source_url,
        title: title
      });
      if (res && res.ok) {
        setProgress('已开始下载完整视频 → 本地 Downloads\\videos（详见 bridge 终端进度）');
        el.state.textContent = '已下发下载';
      } else {
        setProgress('下载请求失败：' + ((res && res.error) || '未知错误'));
        el.state.textContent = '失败';
      }
    } catch (err) {
      setProgress('无法连接本地下载服务：' + err.message + '（请先双击启动 bridge.js）');
      el.state.textContent = '未启动 bridge';
    } finally {
      S.busy = false;
    }
  }

  /* POST 到本地 bridge 服务（127.0.0.1:8321），不写队列、不推远端 */
  function postToBridge(endpoint, body) {
    return new Promise((resolve, reject) => {
      const doSend = () => {
        const onload = r => {
          try { resolve(JSON.parse(r.responseText)); }
          catch (_) { resolve(null); }
        };
        if (typeof GM_xmlhttpRequest === 'function') {
          GM_xmlhttpRequest({
            method: 'POST',
            url: BRIDGE + endpoint,
            headers: { 'Content-Type': 'application/json' },
            data: JSON.stringify(body),
            onload,
            onerror: () => reject(new Error('网络错误')),
            ontimeout: () => reject(new Error('连接超时'))
          });
        } else {
          fetch(BRIDGE + endpoint, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(body)
          }).then(onload).catch(reject);
        }
      };
      doSend();
    });
  }

  /* 撤回：删除刚入队的那一条 pending_imports（走本地 bridge 用 service_role 删） */
  async function undoLast() {
    if (S.busy) return;
    if (!S.lastQueuedId) return setProgress('没有可撤回的入队记录');
    S.busy = true;
    el.state.textContent = '撤回中…';
    try {
      const res = await postToBridge('/delete', { id: S.lastQueuedId });
      if (res && res.ok) {
        setProgress('已撤回 ✓ ' + (res.deleted > 0 ? '该条已从队列移除' : '未找到该条（可能已处理）'));
        el.state.textContent = '已撤回';
        el.undo.style.display = 'none';
        S.lastQueuedId = null;
      } else {
        setProgress('撤回失败：' + ((res && res.error) || '未知错误'));
        el.state.textContent = '失败';
      }
    } catch (err) {
      setProgress('撤回失败：' + err.message + '（请确认 bridge 服务已启动）');
      el.state.textContent = '失败';
    } finally {
      S.busy = false;
    }
  }

  /* 全屏适配：body 下的面板在全屏时不可见（浏览器只渲染全屏元素及其后代）。
     进入全屏 → 把面板搬进全屏元素内部（改 absolute 定位）；退出 → 搬回 body。
     注意：若全屏元素是 <video> 本身（子元素不渲染）则无法显示，微博/腾讯全屏的是容器 div，可正常处理。 */
  (() => {
    const stopper = e => e.stopPropagation();
    // 挡住指针事件冒泡，防止点击面板触发网站播放器的播放/暂停；键盘不挡（否则自己 window 上的热键会失效）
    ['mousedown', 'mouseup', 'click', 'dblclick', 'touchstart', 'contextmenu'].forEach(ev =>
      panel.addEventListener(ev, stopper)
    );
    document.addEventListener('fullscreenchange', () => {
      const fs = document.fullscreenElement;
      if (fs) {
        if (fs.tagName === 'VIDEO' || fs === panel) return;
        fs.appendChild(panel);
        panel.classList.add('wbclip-infs');
      } else {
        document.body.appendChild(panel);
        panel.classList.remove('wbclip-infs');
      }
    });
  })();

  panel.addEventListener('click', e => {
    const btn = e.target.closest('button[data-act]');
    if (!btn) return;
    ({ back5: () => seekBy(-5), a: setA, b: setB, fwd5: () => seekBy(5),
      preview: togglePreview, save: save, full: downloadFull, undo: undoLast, clear: clearAll })[btn.dataset.act]();
  });

  /* 腾讯：起止时间输入实时刷新时长；在输入框里按 Enter 直接入队 */
  el.start.addEventListener('input', refreshTxDur);
  el.end.addEventListener('input', refreshTxDur);
  [el.start, el.end].forEach(inp => {
    inp.addEventListener('keydown', e => {
      if (e.key === 'Enter') { e.preventDefault(); save(); }
    });
  });

  /* 输入框里的空格键会被网站自己的全局快捷键接走（微博/腾讯都用空格切播放/暂停，
     它们在 document 上 preventDefault，导致空格打不进输入框）。
     这里在「事件到达网站之前」把键盘事件截住：只 stopPropagation，不 preventDefault，
     所以输入框自身照常收到空格；面板自己的 window 热键也不会在打字时被误触发。 */
  [el.tags, el.note, el.start, el.end].forEach(inp => {
    ['keydown', 'keypress', 'keyup'].forEach(ev =>
      inp.addEventListener(ev, e => e.stopPropagation())
    );
  });
  /* 标签/备注框里按 Enter 也能直接入队（与起止时间框一致） */
  [el.tags, el.note].forEach(inp => {
    inp.addEventListener('keydown', e => {
      if (e.key === 'Enter') { e.preventDefault(); save(); }
    });
  });

  window.addEventListener('keydown', e => {
    if (e.ctrlKey || e.metaKey || e.altKey) return;
    const t = e.target;
    if (t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA')) return;
    if (e.key === 'i') setA();
    else if (e.key === 'o') setB();
    else if (e.key === 'p') togglePreview();
    else if (e.key === 'Enter') { e.preventDefault(); save(); }
    else if (e.key === 'Escape') clearAll();
  });
