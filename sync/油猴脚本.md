// ==UserScript==
// @name         微博视频片段标记器（tactic-lab 队列版）
// @namespace    https://niuaoyu.github.io/ball
// @version      0.2.0
// @description  在微博视频页标记 I/O 片段，写入 Supabase 待导入队列，电脑端按需下载裁剪（路线 A：省流量 + 帧级精确）。不下载、不录制。
// @author       niuaoyu
// @match        https://weibo.com/tv/show/*
// @match        https://video.weibo.com/show*
// @match        https://weibo.com/*
// @run-at       document-idle
// @grant        GM_xmlhttpRequest
// @connect      xzfnznwovpfovyssolrb.supabase.co
// ==/UserScript==

(function () {
  'use strict';

  const SUPABASE_URL = 'https://xzfnznwovpfovyssolrb.supabase.co';
  const ANON_KEY = 'sb_publishable_NRdVd7YIdp_aJfpG5cH-BQ_MqM1U3US';

  const PANEL_HTML = `
  <div id="wbclip-panel">
    <div class="wbclip-head">微博片段标记 <span id="wbclip-state">等待视频…</span></div>
    <div class="wbclip-row">
      <button data-act="a">设开始 <b>I</b></button>
      <button data-act="b">设结束 <b>O</b></button>
      <button data-act="preview" id="wbclip-preview">预览循环 <b>P</b></button>
      <button data-act="save" class="primary">入队 <b>Enter</b></button>
      <button data-act="clear">清除 <b>Esc</b></button>
    </div>
    <div class="wbclip-time">
      A <span id="wbclip-a">--:--.---</span> &nbsp; B <span id="wbclip-b">--:--.---</span>
      &nbsp; 当前 <span id="wbclip-now">--:--.---</span>
    </div>
    <input id="wbclip-tags" placeholder="标签（空格分隔，默认维度 战术），如：挡拆 下顺">
    <input id="wbclip-note" placeholder="备注（可选）">
    <div id="wbclip-progress"></div>
  </div>`;

  const CSS = `
  #wbclip-panel{position:fixed;right:14px;top:14px;z-index:2147483647;width:330px;
    background:#fff;color:#222;border:1px solid #ddd;border-radius:12px;padding:10px 12px;
    font:13px/1.5 -apple-system,"Segoe UI","PingFang SC","Microsoft YaHei",sans-serif;
    box-shadow:0 8px 28px rgba(0,0,0,.14)}
  #wbclip-panel .wbclip-head{font-weight:700;font-size:13px;margin-bottom:8px;
    display:flex;justify-content:space-between;align-items:center}
  #wbclip-panel .wbclip-head span{font-weight:400;color:#0f766e;font-size:12px}
  #wbclip-panel .wbclip-row{display:flex;gap:6px;flex-wrap:wrap;margin-bottom:8px}
  #wbclip-panel button{font:inherit;font-size:12px;padding:5px 8px;border:1px solid #d8d4ca;
    background:#f5f3ee;border-radius:7px;cursor:pointer;color:#222}
  #wbclip-panel button:hover{background:#ece9e1}
  #wbclip-panel button.primary{background:#0f766e;border-color:#0f766e;color:#fff}
  #wbclip-panel button.on{background:#0f766e;border-color:#0f766e;color:#fff}
  #wbclip-panel button b{font-weight:700;opacity:.75}
  #wbclip-panel .wbclip-time{font-variant-numeric:tabular-nums;color:#555;margin-bottom:8px;font-size:12px}
  #wbclip-panel input{width:100%;font:inherit;padding:6px 9px;border:1px solid #d8d4ca;
    border-radius:7px;margin-bottom:6px;outline:none;background:#faf9f5}
  #wbclip-panel input:focus{border-color:#0f766e;background:#fff}
  #wbclip-progress{font-size:12px;color:#8a857a;min-height:16px;word-break:break-all}
  `;

  const styleEl = document.createElement('style');
  styleEl.textContent = CSS;
  document.head.appendChild(styleEl);
  const box = document.createElement('div');
  box.innerHTML = PANEL_HTML;
  document.body.appendChild(box.firstElementChild);

  const $ = s => document.querySelector(s);
  const panel = $('#wbclip-panel');
  const el = {
    state: $('#wbclip-state'),
    a: $('#wbclip-a'), b: $('#wbclip-b'), now: $('#wbclip-now'),
    tags: $('#wbclip-tags'), note: $('#wbclip-note'),
    progress: $('#wbclip-progress'), preview: $('#wbclip-preview')
  };

  const S = { video: null, A: null, B: null, busy: false, previewing: false };

  function fmt(t) {
    if (t == null || !isFinite(t) || t < 0) return '--:--.---';
    const m = Math.floor(t / 60), s = Math.floor(t % 60), ms = Math.floor((t % 1) * 1000);
    return String(m).padStart(2, '0') + ':' + String(s).padStart(2, '0') + '.' + String(ms).padStart(3, '0');
  }
  const r3 = n => Math.round(n * 1000) / 1000;
  function uid(p) {
    return p + '-' + Date.now().toString(36) + '-' + Math.random().toString(36).slice(2, 8);
  }
  function setProgress(t) { el.progress.textContent = t || ''; }
  function renderTimes() {
    el.a.textContent = fmt(S.A);
    el.b.textContent = fmt(S.B);
    el.now.textContent = S.video ? fmt(S.video.currentTime) : '--:--.---';
  }
  function getFid() {
    const m = location.pathname.match(/show\/([0-9:]+)/) || location.search.match(/fid=([0-9:]+)/);
    return m ? m[1] : '';
  }

  function findVideo() {
    const vids = Array.from(document.querySelectorAll('video'))
      .filter(v => v.duration > 0 && (v.currentSrc || v.src));
    vids.sort((x, y) => (y.videoWidth || 0) - (x.videoWidth || 0));
    return vids[0] || null;
  }
  let waitCount = 0;
  const waitTimer = setInterval(() => {
    const v = findVideo();
    if (v) {
      S.video = v;
      clearInterval(waitTimer);
      el.state.textContent = '就绪 · ' + Math.round(v.duration) + 's';
      setInterval(renderTimes, 120);
    } else if (++waitCount > 120) {
      clearInterval(waitTimer);
      el.state.textContent = '未找到视频';
    }
  }, 800);

  function setA() {
    if (!S.video || S.busy) return;
    stopPreview();
    S.A = S.video.currentTime;
    if (S.B != null && S.B <= S.A) S.B = null;
    renderTimes();
  }
  function setB() {
    if (!S.video || S.busy) return;
    if (S.A == null) return setProgress('先按 I 标记开始点');
    S.B = S.video.currentTime;
    if (S.B <= S.A) { const t = S.A; S.A = S.B; S.B = t; }
    renderTimes();
  }
  function clearAll() {
    stopPreview();
    S.A = null; S.B = null;
    renderTimes(); setProgress('');
  }

  function togglePreview() {
    if (!S.video) return;
    if (S.previewing) return stopPreview();
    if (S.A == null || S.B == null) return setProgress('先标记 A 和 B 才能预览');
    S.video.currentTime = S.A;
    S.video.play();
    S.previewing = true;
    el.preview.classList.add('on');
    setProgress('预览循环中（按 P 停止）');
  }
  function stopPreview() {
    if (!S.previewing) return;
    S.previewing = false;
    el.preview.classList.remove('on');
    setProgress('');
  }
  function previewTick() {
    if (S.previewing && S.video && S.B != null && !S.video.paused) {
      if (S.video.currentTime >= S.B) S.video.currentTime = S.A;
    }
    requestAnimationFrame(previewTick);
  }
  requestAnimationFrame(previewTick);

  function post(body) {
    return new Promise((resolve, reject) => {
      const doPost = () => {
        if (typeof GM_xmlhttpRequest === 'function') {
          GM_xmlhttpRequest({
            method: 'POST',
            url: SUPABASE_URL + '/rest/v1/pending_imports',
            headers: {
              'apikey': ANON_KEY, 'Authorization': 'Bearer ' + ANON_KEY,
              'Content-Type': 'application/json', 'Prefer': 'return=minimal'
            },
            data: JSON.stringify(body),
            onload: r => (r.status >= 200 && r.status < 300) ? resolve() : reject(new Error('HTTP ' + r.status)),
            onerror: () => reject(new Error('网络错误（检查 @connect）'))
          });
        } else {
          fetch(SUPABASE_URL + '/rest/v1/pending_imports', {
            method: 'POST',
            headers: {
              'apikey': ANON_KEY, 'Authorization': 'Bearer ' + ANON_KEY,
              'Content-Type': 'application/json', 'Prefer': 'return=minimal'
            },
            body: JSON.stringify(body)
          }).then(r => r.ok ? resolve() : reject(new Error('HTTP ' + r.status)))
            .catch(reject);
        }
      };
      doPost();
    });
  }

  async function save() {
    if (S.busy) return;
    if (!S.video) return setProgress('视频未就绪');
    if (S.A == null || S.B == null) return setProgress('先用 I / O 标记起止点');
    const A = Math.min(S.A, S.B), B = Math.max(S.A, S.B);
    const fid = getFid();
    if (!fid) return setProgress('未能识别视频 fid');

    S.busy = true;
    el.state.textContent = '入队中…';
    stopPreview();
    try {
      const tagWords = el.tags.value.trim().split(/\s+/).filter(Boolean);
      const tags = tagWords.map(v => ({ dimension: '战术', value: v }));
      const note = el.note.value.trim();
      const id = uid('clip');
      const title = document.title.replace(/\s*-\s*微博.*$/, '').trim();
      await post({
        id, fid, page_url: location.href, title,
        start_sec: r3(A), end_sec: r3(B), tags, note, status: 'pending'
      });
      setProgress('已入队 ✓ 回电脑跑 node import.js --queue 下载裁剪');
      el.state.textContent = '已入队';
      el.tags.value = ''; el.note.value = '';
      clearAll();
    } catch (err) {
      setProgress('入队失败：' + err.message);
      el.state.textContent = '失败';
    } finally {
      S.busy = false;
    }
  }

  panel.addEventListener('click', e => {
    const btn = e.target.closest('button[data-act]');
    if (!btn) return;
    ({ a: setA, b: setB, preview: togglePreview, save: save, clear: clearAll })[btn.dataset.act]();
  });

  window.addEventListener('keydown', e => {
    if (e.ctrlKey || e.metaKey || e.altKey) return;
    const t = e.target;
    if (t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA')) return;
    if (e.key === 'i') setA();
    else if (e.key === 'o') setB();
    else if (e.key === 'p') togglePreview();
    else if (e.key === 'Enter') { e.preventDefault(); save(); }
    else if (e.key === 'Escape') clearAll();
  });
})();
